SQL do Supabase

create table if not exists portfolio_admins (
  email text primary key
);

create table if not exists portfolio_artes (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  category text not null default 'geral',
  image_url text not null,
  project_url text,
  notes text,
  published boolean not null default true,
  sort_order integer not null default 0,
  created_by uuid default auth.uid(),
  created_at timestamptz not null default now()
);

alter table portfolio_admins enable row level security;
alter table portfolio_artes enable row level security;

create policy "admins can read own allowlist"
on portfolio_admins for select
using (email = auth.jwt() ->> 'email');

create policy "public can read published portfolio"
on portfolio_artes for select
using (published = true);

create policy "admins can read all portfolio"
on portfolio_artes for select
using (exists (
  select 1 from portfolio_admins
  where email = auth.jwt() ->> 'email'
));

create policy "admins can insert portfolio"
on portfolio_artes for insert
with check (exists (
  select 1 from portfolio_admins
  where email = auth.jwt() ->> 'email'
));

create policy "admins can update portfolio"
on portfolio_artes for update
using (exists (
  select 1 from portfolio_admins
  where email = auth.jwt() ->> 'email'
))
with check (exists (
  select 1 from portfolio_admins
  where email = auth.jwt() ->> 'email'
));

create policy "admins can delete portfolio"
on portfolio_artes for delete
using (exists (
  select 1 from portfolio_admins
  where email = auth.jwt() ->> 'email'
));

insert into portfolio_admins (email)
values ('SEU_EMAIL_AQUI')
on conflict (email) do nothing;

insert into storage.buckets (id, name, public)
values ('portfolio-artes', 'portfolio-artes', true)
on conflict (id) do nothing;

create policy "public can read portfolio files"
on storage.objects for select
using (bucket_id = 'portfolio-artes');

create policy "admins can upload portfolio files"
on storage.objects for insert
with check (
  bucket_id = 'portfolio-artes'
  and exists (
    select 1 from portfolio_admins
    where email = auth.jwt() ->> 'email'
  )
);
